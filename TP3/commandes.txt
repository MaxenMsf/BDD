db.Restau.aggregate([
  {
    $group: {
      _id: { borough: "$borough", cuisine: "$cuisine" }
    }
  },
  {
    $project: {
      _id: 0,
      borough: "$_id.borough",
      cuisine: "$_id.cuisine"
    }
  }
]) = Lister toutes les cuisines uniques et leurs arrondissements respectifs. (L’opérateur $project dans une agrégation MongoDB permet de sélectionner ou transformer les champs qui seront présents dans le résultat final. Exemple : $_id.cuisine devient cuisine)

db.Restau.aggregate([
  { $unwind: "$grades" },
  { $group: {
      _id: "$name",
      average_score: { $avg: "$grades.score" }
    }
  },
  { $sort: { average_score: -1 } },
  { $limit: 5 }
]) = Trouver les 5 meilleurs restaurants avec le score d’inspection moyen le plus élevé.

db.Restau.aggregate([
  { $unwind: "$grades" },
  { $match: { "grades.score": { $ne: null } } },
  { $group: {
      _id: "$name",
      min_score: { $min: "$grades.score" }
    }
  },
  { $sort: { min_score: 1 } },
  { $limit: 1 }
]) = Trouver le restaurant avec le score d’inspection minimum ($ne = not equal)
